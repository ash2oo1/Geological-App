// --- Main Contour Generation Engine (Professional Isolines) --- function processDataAndRender() { if (globalPoints.length < 4) return; const loader = document.getElementById('loader'); loader.classList.add('active'); setTimeout(() => { try { if (geojsonLayer) map.removeLayer(geojsonLayer); let minZ = Infinity; let maxZ = -Infinity; // 1. تحويل النقاط المرفوعة let features = []; globalPoints.forEach(pt => { let lon = pt[0]; let lat = pt[1]; let z = pt[2]; features.push(turf.point([lon, lat], { z: z })); if (z > maxZ) maxZ = z; if (z < minZ) minZ = z; }); const pointsCollection = turf.featureCollection(features); // 2. إنشاء الغلاف الخارجي (Convex Hull) لقص الخريطة عليه ومنع المربع الوهمي const hull = turf.convex(pointsCollection); // 3. إنشاء شبكة مساحية دقيقة جداً (Grid) const resolution = parseFloat(document.getElementById('mapResolution').value); // استخدام وزن 1.5 لتقليل ظاهرة "عين الثور" const grid = turf.interpolate(pointsCollection, resolution, { gridType: 'point', property: 'z', units: 'kilometers', weight: 1.5 }); // 4. تحديد المسافة الكنتورية (Contour Interval) const interval = parseFloat(document.getElementById('contourInterval').value) || 10; let breaks = []; let startBreak = Math.floor(minZ / interval) * interval; let endBreak = Math.ceil(maxZ / interval) * interval; for (let i = startBreak; i <= endBreak; i += interval) { breaks.push(i); } // 5. رسم خطوط الكنتور الحقيقية (Isolines) بدلاً من المضلعات (Isobands) const lines = turf.isolines(grid, breaks, { zProperty: 'z' }); // 6. عرض الخطوط على الخريطة geojsonLayer = L.geoJSON(lines, { style: function (feature) { let zValue = feature.properties.z; let color = getColorForZ(zValue, minZ, maxZ); // تمييز الخطوط الرئيسية (Index Contours) بخط أسمك let isIndex = (zValue % (interval * 5) === 0); return { color: color, weight: isIndex ? 3 : 1.5, // خطوط عريضة للكنتور الرئيسي opacity: 0.9, className: 'contour-line' }; }, onEachFeature: function (feature, layer) { // وضع رقم الارتفاع على الخط نفسه layer.bindTooltip(`${feature.properties.z}`, { permanent: true, direction: 'center', className: 'contour-text-label' }); } }).addTo(map); // إضافة ستايل CSS للارقام لتظهر فوق الخطوط بشفافية let style = document.createElement('style'); style.innerHTML = ` .contour-text-label { background: rgba(20, 27, 45, 0.7); border: none; box-shadow: none; color: white; font-weight: bold; font-size: 10px; padding: 1px 3px; border-radius: 4px; } .leaflet-tooltip-left::before, .leaflet-tooltip-right::before { display: none; } `; document.head.appendChild(style); // 7. توجيه الخريطة map.fitBounds(geojsonLayer.getBounds(), {padding: [30, 30]}); document.getElementById('lblMin').innerText = `${minZ.toFixed(1)}m`; document.getElementById('lblMid').innerText = `${((minZ + maxZ) / 2).toFixed(1)}m`; document.getElementById('lblMax').innerText = `${maxZ.toFixed(1)}m`; document.getElementById('colorLegend').style.display = 'flex'; } catch (error) { console.error(error); alert("خطأ في المعالجة. تأكد من أن النقاط المرفوعة في ملف CSV تغطي مساحة كافية لتوليد خطوط كنتورية."); } finally { loader.classList.remove('active'); } }, 100); }
